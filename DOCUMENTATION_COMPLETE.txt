I. VUE D'ENSEMBLE DU PROJET
================================================================================

Contexte : Projet réalisé en groupe dans le cadre d'une mission simulée pour l'OMS
Objectif : Mise en place d'une chaîne complète de traitement, d'analyse et de 
visualisation de données sanitaires (COVID-19 et Mpox)
Technologies principales : Python, FastAPI, React, PostgreSQL, Docker, i18n

II. ARCHITECTURE TECHNIQUE
================================================================================

2.1 Modèle de données (UML)

Le projet suit une architecture en étoile typique des entrepôts de données :

d_location (Dimension géographique)
    ├── location_id (PK)
    └── location_name
    
f_covid (Faits COVID-19)
    ├── covid_fact_id (PK)
    ├── date
    ├── location_id (FK → d_location)
    ├── total_cases, new_cases
    ├── total_deaths, new_deaths
    ├── icu_patients, hosp_patients
    └── total_vaccinations, people_vaccinated

f_mpox (Faits Mpox)
    ├── mpox_fact_id (PK)
    ├── date
    ├── location_id (FK → d_location)
    ├── total_cases, new_cases
    ├── total_deaths, new_deaths
    └── métriques per million

f_predi_covid (Prédictions)
    ├── pred_id (PK)
    ├── date_predite
    ├── location_id (FK → d_location)
    ├── indicateur
    └── valeur_predite

users (Authentification)
    ├── user_id (PK)
    ├── username, email
    └── password_hash

3.1 Modèles SQLAlchemy

Code : backend/app/models/models.py

from sqlalchemy import Column, Integer, String, Date, DateTime, Numeric, ForeignKey, func, Boolean
from app.core.database import Base

class DLocation(Base):
    """Dimension géographique pour stocker les localisations uniques"""
    __tablename__ = 'd_location'
    
    location_id = Column(Integer, primary_key=True, autoincrement=True)
    location_name = Column(String(100), nullable=False, unique=True)  

class FCovid(Base):
    """Table de faits principale pour les données COVID"""
    __tablename__ = 'f_covid'
    
    covid_fact_id = Column(Integer, primary_key=True, autoincrement=True)
    date = Column(Date, nullable=False)
    location_id = Column(Integer, ForeignKey('d_location.location_id'), nullable=False)
    
    # Métriques de cas et décès
    total_cases = Column(Numeric(15, 2))
    new_cases = Column(Numeric(15, 2))
    total_deaths = Column(Numeric(15, 2))
    new_deaths = Column(Numeric(15, 2))
    
    # Métriques hospitalières
    icu_patients = Column(Numeric(15, 2))
    hosp_patients = Column(Numeric(15, 2))
    
    # Métriques de vaccination
    total_vaccinations = Column(Numeric(15, 2))
    people_vaccinated = Column(Numeric(15, 2))
    
    # Métadonnées
    created_at = Column(DateTime, server_default=func.now())
    updated_at = Column(DateTime, server_default=func.now(), onupdate=func.now())

class CountryConfig:
    def __init__(self, technical_api: bool = True, rgpd: bool = False, 
                 multi_language: bool = False, languages: List[str] = None):
        self.technical_api_enabled = technical_api
        self.rgpd_mode = rgpd
        self.multi_language = multi_language
        self.supported_languages = languages or ["en"]

class Settings:
    # Configuration fixe par pays
    COUNTRY_CONFIGS: Dict[str, CountryConfig] = {
        "usa": CountryConfig(
            technical_api=True,      # API technique activée
            rgpd=False,              # Pas de contraintes RGPD
            multi_language=False,    # Anglais uniquement
            languages=["en"]
        ),
        "france": CountryConfig(
            technical_api=False,     # API technique désactivée
            rgpd=True,               # Contraintes RGPD strictes

            languages=["fr"]
        ),
        "switzerland": CountryConfig(
            technical_api=False,     # API technique désactivée
            rgpd=False,              # Contraintes RGPD modérées
            multi_language=True,     # Multilingue
            languages=["fr", "de", "it"]
        )
    }

3.3 Endpoints API avec RGPD

Code : backend/app/api/endpoints/locations.py

from app.api.dependencies import check_rgpd_compliance

@router.get("/", response_model=List[DLocationRead])
@check_rgpd_compliance()  # Vérification RGPD automatique
async def liste_des_pays_fc(
    skip: int = 0, 
    limit: int = 100, 
    db: AsyncSession = Depends(get_db)
):
    locations = await liste_pays(db, skip=skip, limit=limit)
    return locations

@router.post("/", response_model=DLocationRead, status_code=status.HTTP_201_CREATED)
@check_rgpd_compliance()
async def creer_pays_fc(
    location: DLocationCreate, 
    db: AsyncSession = Depends(get_db)
):
    # Vérifier si la location existe déjà
    db_location = await obtenir_pays_par_nom(db, location.location_name)
    if db_location:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Le pays '{location.location_name}' existe déjà"
        )
    
    return await creer_pays(db, location)

const getEnvVars = () => {
  // Validation des variables requises

    build:
      context: .
      dockerfile: backend/Dockerfile
    expose:
      - "8000"
    depends_on:

        condition: service_healthy
    environment:
      - COUNTRY=${COUNTRY}  # Configuration pays
      - DATABASE_URL=${DATABASE_URL}
      - SECRET_KEY=${SECRET_KEY}
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/health"]
      interval: 10s

5.2 Configuration Nginx (Reverse Proxy)

Code : nginx.conf

server {
    listen 80;
    client_max_body_size 200M;

    # Frontend React
    location / {
        proxy_pass http://frontend/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }

    # Backend API FastAPI
    location /api/ {
        proxy_pass http://backend:8000/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # Documentation Swagger
    location /docs {
        proxy_pass http://backend:8000/docs;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # PgAdmin (optionnel)
    location /pgadmin/ {
        proxy_pass http://pgadmin:80/;
        proxy_set_header X-Script-Name /pgadmin;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

6.1 Initialisation de la base de données

Code : backend/scripts/init_db.py

import asyncio
from app.core.database import Base, engine
from app.models import models

async def init_db():
    """Crée toutes les tables de la base de données"""
    try:
        # Créer les tables
        async with engine.begin() as conn:
            await conn.run_sync(Base.metadata.create_all)
        print("✅ Création des tables terminée.")
            
    except Exception as e:
        print(f"❌ Erreur lors de l'initialisation de la base : {e}")

if __name__ == "__main__":
    try:
        asyncio.run(init_db())
    except Exception as e:
        print(f"❌ Erreur lors de l'initialisation de la base : {e}")
